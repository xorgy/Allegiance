CMAKE_MINIMUM_REQUIRED(VERSION 3.24)
CMAKE_POLICY(SET CMP0005 NEW)

PROJECT(Allegiance)

OPTION(ALLEGIANCE_NATIVE_CLIENT "Build the portable Vulkan client and authority" ${CMAKE_HOST_UNIX})
IF(ALLEGIANCE_NATIVE_CLIENT)
    ENABLE_TESTING()
    ADD_SUBDIRECTORY(src/native)
    RETURN()
ENDIF()

OPTION(USE_STEAM "Use steam" OFF)

MACRO(SET_MODULE MODULE_NAME)
    ADD_DEFINITIONS(-D__MODULE__="${MODULE_NAME}")
    INCLUDE_DIRECTORIES(../Inc)
    IF (USE_STEAM)
        INCLUDE_DIRECTORIES(../Lib/steam)
    ELSE()
        ADD_DEFINITIONS(-DNO_STEAM)
    ENDIF()
ENDMACRO()


ADD_SUBDIRECTORY(src)
